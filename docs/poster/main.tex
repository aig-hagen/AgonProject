\begin{columns}\column{1}\block{}{\sectionrule\vspace{0.6em}\eyebrow{Features}}\end{columns}
\begin{columns}
\column{0.2}\block{}{\featureitem{Construct}{\faProjectDiagram}{Interactive graph editor with automatic layouts and physics simulation.}}
\column{0.2}\block{}{\featureitem{Evaluate}{\faBalanceScale}{Extension- and ranking-based semantics, with enumeration and credulous / skeptical acceptance.}}
\column{0.2}\block{}{\featureitem{Export \& share}{\faShareSquare}{Export to \LaTeX, SVG, TGF, and the ICCMA AF format, or share by URL.}}
\column{0.2}\block{}{\featureitem{Generate}{\faDice}{Random frameworks from \stat{8} configurable graph generators.}}
\column{0.2}\block{}{\featureitem{Learn}{\faGraduationCap}{\stat{10+} tutorials and a glossary with definitions and literature references.}}
\end{columns}

\begin{columns}
\column{0.52}
\block{}{
    \sectionrule\vspace{0.6em}
    \eyebrow{Supported semantics}
    \semrow{Classical}{conflict-free, admissible, complete, grounded, preferred, stable}
    \semrow{Admissibility-based}{ideal, semi-stable, eager, strong admissibility, initial sets, \dots}
    \semrow{Non-admissible}{naive, CF2, SCF2, stage, stage2, (strongly) undisputed}
    \semrow{Meta-families}{weak admissibility, (semi-)qualified, vacuous-reduct, serialisable}
    \semrow{Argument-rankings}{burden, categoriser, counting, discussion, iterated graded defense, probabilistic, propagation, serialisability, strategy-based, \dots}
}
\column{0.06}
\column{0.42}
\block{}{
    \sectionrule\vspace{0.6em}
    \eyebrow{Outlook}
    \vspace{0.7em}
    {\color{inksoft}\fontsize{25}{33}\selectfont
    \begin{itemize}\setlength{\itemsep}{0.4em}
        \item more formalisms, e.\,g., \textbf{ABA}, \textbf{weighted AFs}, \textbf{BSAFs}, \textbf{claim-augmented AFs}
        \item benchmark generators from the ICCMA ecosystem
        \item other reasoning approaches, e.\,g., acceptance explanations
    \end{itemize}}
    \vspace{0.4em}
    \callout{{\sc\agon} is backend-agnostic and open to new formalisms and solvers.}
